type TextProps = HTMLAttributes<HTMLElement> & {
  as?: "h1" | "h2" | "h3" | "p" | "span" | "div"
  variant?: "hero" | "page" | "section" | "card" | "body" | "small" | "meta" | "eyebrow"
  children: ReactNode
}

type LinkProps = Omit<ComponentProps<typeof Link>, "href" | "as"> & {
  to: ComponentProps<typeof Link>["href"]
}

// Astra has no semantic typography or general-purpose blog surface primitive.
export function Typography({
  as = "p",
  variant = "body",
  className = "",
  children,
  ...props
}: TextProps) {
  return createElement(
    as,
    { ...props, className: `type-${variant} ${className}` },
    children,
  )
}

export function GlassSurface({
  as = "div",
  variant = "glass",
  className = "",
  children,
  ...props
}: HTMLAttributes<HTMLElement> & {
  as?: "div" | "section" | "article" | "aside"
  variant?: "glass" | "reading"
  children: ReactNode
}) {
  return createElement(
    as,
    {
      ...props,
      className: `glass blog-surface blog-surface--${variant} ${className}`,
    },
    children,
  )
}

export function AppLink({ to, className = "", children, ...props }: LinkProps) {
  return (
    <Link href={to} {...props} className={`app-link ${className}`}>
      {children}
    </Link>
  )
}

export function TextLink({ children, ...props }: LinkProps) {
  return (
    <AppLink {...props} className={`text-link ${props.className || ""}`}>
      {children}
      <ArrowRight className="icon-sm" aria-hidden="true" />
    </AppLink>
  )
}

export function ExternalLink({
  href,
  children,
  className = "",
  ...props
}: Omit<LinkProps, "to"> & { href: string }) {
  if (!/^(https:\/\/|mailto:)/.test(href)) return <span>{children}</span>
  return (
    <Link
      href={href}
      target={href.startsWith("mailto:") ? undefined : "_blank"}
      rel="noopener noreferrer"
      className={`app-link ${className}`}
      {...props}
    >
      {children}
    </Link>
  )
}

export function SectionHeading({
  title,
  eyebrow,
  children,
}: {
  title: string
  eyebrow?: string
  children?: ReactNode

        {eyebrow && <Typography variant="eyebrow">{eyebrow}</Typography>}
        <Typography as="h2" variant="section">
          {title}
        </Typography>
      </div>
      {children}
    </div>
  )
}

export const inlineText = (content: Inline[]) =>
  content.map((item) => (typeof item === "string" ? item : item.text)).join("")

export function blockText(block: ArticleBlock): string {
  switch (block.kind) {
    case "heading":
      return block.text
    case "paragraph":
    case "quote":
      return inlineText(block.content)
    case "code":
      return block.code
    case "list":
      return block.items.map(inlineText).join(" ")
    case "table":
      return [...block.headers, ...block.rows.flat()].join(" ")
  }
}

export function readingMinutes(article: Article) {
  const text = article.blocks.map(blockText).join(" ")
  const chinese = (text.match(/[\u3400-\u9fff]/g) || []).length
  const words = (text.match(/[a-zA-Z0-9]+/g) || []).length
  return Math.max(1, Math.ceil(chinese / 280 + words / 200))
}

export function formatDate(date: string, compact = false) {
  const [year, month, day] = date.split("-")
  return compact
    ? `${month}.${day}`
    : `${year} 年 ${Number(month)} 月 ${Number(day)} 日`
}

export function searchArticles(query: string, category = "") {
  const normalized = query.trim().toLocaleLowerCase()
  return articles.filter(
    (article) =>
      (!category || article.category === category) &&
      `${article.title} ${article.excerpt} ${article.tags.join(" ")}`
        .toLocaleLowerCase()
        .includes(normalized),
  )
}

export function getBlogState(params: URLSearchParams) {
  const query = params.get("q") || ""
  const rawCategory = params.get("category") || ""
  const category = categories.some((item) => item.id === rawCategory)
    ? rawCategory
    : ""
  const results = searchArticles(query, category)
  const pageCount = Math.max(1, Math.ceil(results.length / 6))
  const rawPage = params.get("page") || "1"
  const parsedPage = /^[1-9]\d*$/.test(rawPage) ? Number(rawPage) : 1
  const page = Math.min(
    Number.isSafeInteger(parsedPage) ? parsedPage : 1,
    pageCount,
  )
  return {
    query,
    category,
    results,
    pageCount,
    page,
    visible: results.slice((page - 1) * 6, page * 6),
  }
}

export function blogParams(query: string, category: string, page = 1) {
  const params = new URLSearchParams()
  if (query.trim()) params.set("q", query)
  if (category) params.set("category", category)
  if (page > 1) params.set("page", String(page))
  return params
}

export function articleHeadings(article: Article) {
  return article.blocks.filter(
    (block): block is Extract<ArticleBlock, { kind: "heading" }> =>
      block.kind === "heading",
  )
}
